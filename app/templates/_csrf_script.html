async function configurationToken(action, signal = AbortSignal.timeout(10000)) { const response = await fetch('/api/v1/auth/csrf-tokens', { method: 'POST', credentials: 'same-origin', redirect: 'error', cache: 'no-store', signal, headers: {'Content-Type': 'application/json', 'X-CampusClaw-Request': 'same-origin'}, body: JSON.stringify({action}), }); const result = await response.json(); if (!response.ok || result.code !== 0) throw new Error(result.message || '无法取得配置写入令牌，请重试。'); return result.data.token; }
